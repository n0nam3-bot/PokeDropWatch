function doGet(e) {
  const page = String(e && e.parameter && e.parameter.page || '').trim().toLowerCase();
  if (page === 'admin') {
    // Admin.html calls google.script.run directly. ALLOWALL lets the GitHub Pages
    // admin shell embed this page, keeping the browser address on GitHub Pages.
    const adminFile = HtmlService.createHtmlOutputFromFile('Admin');
    const fontCss = `
      <style>
        @font-face {
          font-family: "PKMN RBYGSC";
          src: url("https://n0nam3-bot.github.io/PokeDropWatch/assets/fonts/PKMN-RBYGSC.ttf") format("truetype");
          font-style: normal;
          font-weight: 400;
          font-display: swap;
        }
        html, body, body *, body *::before, body *::after, body *::placeholder {
          font-family: "PKMN RBYGSC", "Courier New", monospace !important;
        }
      </style>
    `;
    const themedContent = adminFile.getContent().replace('</head>', fontCss + '</head>');
    return HtmlService.createHtmlOutput(themedContent)
      .setTitle('PokéDrop Watch Admin')
      .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
  }

  const props = PropertiesService.getScriptProperties();
  const template = HtmlService.createTemplateFromFile('Index');
  template.siteOrigin = props.getProperty('SITE_ORIGIN') || 'https://n0nam3-bot.github.io';
  return template.evaluate()
    .setTitle('PokéDrop Watch Verification')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}
