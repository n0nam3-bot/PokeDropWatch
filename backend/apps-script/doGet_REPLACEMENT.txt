Replace only the existing doGet() function in Code.gs with the function below,
then add the pokedropEmbeddedOutput_ helper below it. Keep all other server
functions, Script Properties, and project settings unchanged.

The Admin and Subscribe HTML pages are embedded inside the GitHub Pages site.
The outer browser address therefore remains on n0nam3-bot.github.io. ALLOWALL
is required for these two Apps Script pages to render inside the site iframes.

function doGet(e) {
  const page = String(e && e.parameter && e.parameter.page || '').trim().toLowerCase();

  if (page === 'admin') {
    return pokedropEmbeddedOutput_(HtmlService.createHtmlOutputFromFile('Admin'), 'PokéDrop Watch Admin');
  }

  if (page === 'subscribe' || page === 'notify') {
    return pokedropEmbeddedOutput_(HtmlService.createHtmlOutputFromFile('Subscribe'), 'PokéDrop Watch Email Alerts');
  }

  const props = PropertiesService.getScriptProperties();
  const template = HtmlService.createTemplateFromFile('Index');
  template.siteOrigin = props.getProperty('SITE_ORIGIN') || 'https://n0nam3-bot.github.io';
  return template.evaluate()
    .setTitle('PokéDrop Watch Verification')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

function pokedropEmbeddedOutput_(sourceOutput, title) {
  const fontCss = `
    <style>
      @font-face {
        font-family: "PKMN RBYGSC";
        src: url("https://n0nam3-bot.github.io/PokeDropWatch/assets/fonts/PKMN-RBYGSC.ttf") format("truetype");
        font-style: normal;
        font-weight: 400;
        font-display: swap;
      }
      html, body, body *, body *::before, body *::after, body *::placeholder {
        font-family: "PKMN RBYGSC", "Courier New", monospace !important;
      }
    </style>
  `;

  let html = sourceOutput.getContent();
  // Protect against older Admin.html copies that still contain their own shell.
  html = html.replace(/<header class="top">[\s\S]*?<\/header>/i, '');
  html = html.replace('<p class="foot">PokéDrop Watch · Administrator sessions expire after one hour.</p>',
                      '<p class="foot">Administrator sessions expire after one hour.</p>');
  html = html.replace('</head>', fontCss + '</head>');

  return HtmlService.createHtmlOutput(html)
    .setTitle(title)
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

After saving, deploy the existing web app as a New version. Keep the existing
web app URL and its execution identity/access setting. For an iframe used by a
public GitHub Pages site, the web app must execute as the owner and be available
to Anyone; do not choose an access option that forces visitors to sign in.
